Задание 9. Количество сотрудников и средняя зарплата по отделам
SELECT
    d.DepartmentName,
    COUNT(e.EmployeeId) AS EmployeeCount,
    AVG(e.Salary) AS AverageSalary
FROM Departments d
LEFT JOIN Employees e
    ON d.DepartmentId = e.DepartmentId
GROUP BY d.DepartmentName;
GO

Задание 10. Отделы, в которых средняя зарплата больше 70000
SELECT
    d.DepartmentName,
    AVG(e.Salary) AS AverageSalary
FROM Departments d
JOIN Employees e
    ON d.DepartmentId = e.DepartmentId
GROUP BY d.DepartmentName
HAVING AVG(e.Salary) > 70000;
GO

Задание 22. Создать представление с данными о сотрудниках
CREATE OR ALTER VIEW vw_EmployeeDetails
AS
SELECT
    e.EmployeeId,
    e.EmployeeName,
    e.Salary,
    e.HireDate,
    d.DepartmentName,
    m.EmployeeName AS ManagerName
FROM Employees e
JOIN Departments d
    ON e.DepartmentId = d.DepartmentId
LEFT JOIN Employees m
    ON e.ManagerId = m.EmployeeId;
GO

SELECT *
FROM vw_EmployeeDetails;
GO

Задание 25. Процедура с OUTPUT: средняя зарплата отдела
CREATE OR ALTER PROCEDURE sp_GetAverageSalary
    @DepartmentId INT,
    @AverageSalary DECIMAL(10,2) OUTPUT
AS
BEGIN
    SELECT @AverageSalary = AVG(Salary)
    FROM Employees
    WHERE DepartmentId = @DepartmentId;
END;
GO

DECLARE @Result DECIMAL(10,2);

EXEC sp_GetAverageSalary
    @DepartmentId = 1,
    @AverageSalary = @Result OUTPUT;

SELECT @Result AS AverageSalary;
GO

Задание 26. Функции: скалярная и табличная
CREATE OR ALTER FUNCTION dbo.fn_GetYearlySalary
(
    @MonthlySalary DECIMAL(10,2)
)
RETURNS DECIMAL(12,2)
AS
BEGIN
    RETURN @MonthlySalary * 12;
END;
GO

SELECT
    EmployeeName,
    Salary,
    dbo.fn_GetYearlySalary(Salary) AS YearlySalary
FROM Employees;
GO

(
    @DepartmentId INT
)
RETURNS TABLE
AS

    FROM Employees
    WHERE DepartmentId = @DepartmentId
);
GO

Задание 27. Оконные функции, CTE и CASE
SELECT
    EmployeeName,
    Salary,
    ROW_NUMBER() OVER (ORDER BY Salary DESC) AS RowNumber,
    RANK() OVER (ORDER BY Salary DESC) AS SalaryRank,
    DENSE_RANK() OVER (ORDER BY Salary DESC) AS DenseSalaryRank
FROM Employees;
GO

SELECT
    EmployeeName,
    DepartmentId,
    Salary,
    SUM(Salary) OVER (PARTITION BY DepartmentId) AS DepartmentTotal,
    AVG(Salary) OVER (PARTITION BY DepartmentId) AS DepartmentAverage
FROM Employees;
GO

WITH DepartmentStats AS
(
    SELECT
        DepartmentId,
        COUNT(*) AS EmployeeCount,
        AVG(Salary) AS AverageSalary
    FROM Employees
    GROUP BY DepartmentId
)
SELECT
    d.DepartmentName,
    ds.EmployeeCount,
    ds.AverageSalary
FROM DepartmentStats ds
JOIN Departments d
    ON ds.DepartmentId = d.DepartmentId;
GO

SELECT
    EmployeeName,
    Salary,
    CASE
        WHEN Salary >= 100000 THEN N'Высокая'
        WHEN Salary >= 70000 THEN N'Средняя'
        ELSE N'Начальная'
    END AS SalaryLevel,
    CASE
        WHEN DATEDIFF(YEAR, HireDate, GETDATE()) >= 5
            THEN N'Опытный сотрудник'
        ELSE N'Новый сотрудник'
    END AS Experience
FROM Employees;
GO

SELECT
    EmployeeName,
    HireDate,
    DATEADD(YEAR, 1, HireDate) AS NextWorkAnniversary,
    DATEDIFF(YEAR, HireDate, GETDATE()) AS YearsWorked,
    FORMAT(HireDate, 'dd.MM.yyyy') AS FormattedHireDate
FROM Employees;
GO
